**Goal:** ACP publishes a sanitised JSON of public plans to astrowithroro.com/live, and a static page there renders it.

**Architecture:** A new root module `shooting_publish.py` builds the document from plans, manifest and gear, matching plans to logged targets by sky position. `app.py` gains two thin endpoints over it plus three validated plan fields. A CLI script wraps rescan, build and rsync. The page lives in the separate site repo.

**Tech stack:** Python 3.10+, Flask, unittest, rsync over SSH. Vanilla HTML and JS for the page.

**Spec:** `docs/specs/shooting-page.md`

## Global constraints

- Only plans with `visibility == "public"` ever leave the machine.
- Output must pass `validate_no_paths` from `scripts/sanitise_manifest.py`.
- Coordinates rounded to 0.01 deg, hours to 0.1, dates are `YYYY-MM-DD` strings.
- Tests run with `.venv/bin/python -m pytest tests/ -q` and must stay green.
- No em dashes in prose, comments or commit messages.

```python
"""Build and publish the public "live" document for astrowithroro.com/live.

Spec: docs/specs/shooting-page.md. Only plans with visibility == "public"
are included. The document is scanned for path-shaped strings before it
leaves this module.
"""
from __future__ import annotations

import math
import os
import shlex
import subprocess
import sys
from datetime import date, datetime, timezone
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parent
_SCRIPTS = str(REPO_ROOT / "scripts")
if _SCRIPTS not in sys.path:
    sys.path.insert(0, _SCRIPTS)

from build_archive_manifest import canon_filter  # noqa: E402
from sanitise_manifest import _aperture_class, validate_no_paths  # noqa: E402

DOC_VERSION = 1


def _fov_arcmin(telescope: dict | None, camera: dict | None) -> tuple[float, float]:
    """Same formula as app._fov_arcmin, duplicated so this module has no Flask import."""
    if not telescope or not camera:
        return (0.0, 0.0)
    try:
        fl = float(telescope["focal_length_mm"])
        px_um = float(camera["pixel_size_um"])
        w_px, h_px = camera["sensor_px"]
    except (KeyError, ValueError, TypeError):
        return (0.0, 0.0)
    if fl <= 0:
        return (0.0, 0.0)
    aspp = 206.265 * px_um / fl
    return (w_px * aspp / 60.0, h_px * aspp / 60.0)


def _mosaic_extent_arcmin(fov_w: float, fov_h: float, mosaic: dict | None) -> tuple[float, float]:
    m = mosaic or {}
    rows = max(1, int(m.get("rows") or 1))
    cols = max(1, int(m.get("cols") or 1))
    ov = max(0.0, min(0.99, float(m.get("overlap_pct") or 0) / 100.0))
    return (fov_w * (1 + (cols - 1) * (1 - ov)), fov_h * (1 + (rows - 1) * (1 - ov)))


def _sep_deg(ra1: float, dec1: float, ra2: float, dec2: float) -> float:
    r1, d1, r2, d2 = map(math.radians, (ra1, dec1, ra2, dec2))
    a = math.sin((d2 - d1) / 2) ** 2 + math.cos(d1) * math.cos(d2) * math.sin((r2 - r1) / 2) ** 2
    return math.degrees(2 * math.asin(min(1.0, math.sqrt(a))))


def _diag_arcmin(fov) -> float:
    try:
        w, h = float(fov[0]), float(fov[1])
    except (TypeError, ValueError, IndexError):
        return 0.0
    return math.hypot(w, h)


def _matched_targets(plan_ra: float, plan_dec: float, plan_extent: tuple[float, float], targets: list[dict]) -> list[dict]:
    plan_diag = _diag_arcmin(plan_extent)
    out = []
    for t in targets:
        try:
            ra, dec = float(t["center_ra_deg"]), float(t["center_dec_deg"])
        except (KeyError, TypeError, ValueError):
            continue
        limit = max(plan_diag, _diag_arcmin(t.get("fov_arcmin") or (0, 0))) / 2.0 / 60.0
        if limit > 0 and _sep_deg(plan_ra, plan_dec, ra, dec) <= limit:
            out.append(t)
    return out


def _hours_by_filter(targets: list[dict]) -> dict[str, float]:
    out: dict[str, float] = {}
    for t in targets:
        for fname, cfg in (t.get("filters") or {}).items():
            key = canon_filter(fname) or fname
            out[key] = out.get(key, 0.0) + float((cfg or {}).get("total_hours") or 0)
    return out


def _last_imaged(targets: list[dict]) -> str | None:
    dates = []
    for t in targets:
        dr = t.get("date_range") or []
        if len(dr) == 2 and dr[1]:
            dates.append(str(dr[1])[:10])
    return max(dates) if dates else None


def _project_entry(plan: dict, manifest_targets: list[dict], gear: dict, today: date) -> dict:
    tg = plan.get("target") or {}
    ra = float(tg.get("center_ra_deg") or 0) % 360.0
    dec = float(tg.get("center_dec_deg") or 0)
    tels = {t["id"]: t for t in gear.get("telescopes", [])}
    cams = {c["id"]: c for c in gear.get("cameras", [])}
    telescope = tels.get(plan.get("telescope_id") or "")
    camera = cams.get(plan.get("camera_id") or "")
    fov_w, fov_h = _fov_arcmin(telescope, camera)
    mosaic = tg.get("mosaic") or {}
    ext = _mosaic_extent_arcmin(fov_w, fov_h, mosaic)
    matched = _matched_targets(ra, dec, ext, manifest_targets)
    logged = _hours_by_filter(matched)

    filters = {}
    for fname, goal in (plan.get("filter_goals") or {}).items():
        key = canon_filter(fname) or fname
        done = max(logged.get(key, 0.0), float((goal or {}).get("actual_hours") or 0))
        filters[key] = {
            "target_hours": round(float((goal or {}).get("target_hours") or 0), 1),
            "done_hours": round(done, 1),
        }

    last = _last_imaged(matched)
    nights = (today - date.fromisoformat(last)).days if last else None

    entry = {
        "project_name": str(plan.get("project_name") or ""),
        "target_name": str(tg.get("name") or ""),
        "blurb": str(plan.get("public_blurb") or ""),
        "is_current": bool(plan.get("is_current")),
        "center_ra_deg": round(ra, 2),
        "center_dec_deg": round(dec, 2),
        "fov_arcmin": [round(ext[0], 1), round(ext[1], 1)],
        "telescope": _aperture_class((telescope or {}).get("name") or ""),
        "filters": filters,
        "last_imaged": last,
        "last_imaged_nights_ago": nights,
    }
    rows, cols = int(mosaic.get("rows") or 1), int(mosaic.get("cols") or 1)
    if rows > 1 or cols > 1:
        entry["mosaic"] = {"rows": rows, "cols": cols}
    return entry


def build_shooting_document(plans: list[dict], manifest: dict | None, gear: dict, now: datetime | None = None) -> dict:
    now = now or datetime.now(timezone.utc).astimezone()
    today = now.date()
    targets = list((manifest or {}).get("targets") or [])
    scan = str((manifest or {}).get("scan_date") or "")[:10] or None
    projects = [_project_entry(p, targets, gear, today) for p in plans if p.get("visibility") == "public"]
    projects.sort(key=lambda p: (p["last_imaged"] is None, -(date.fromisoformat(p["last_imaged"]).toordinal()) if p["last_imaged"] else 0, p["project_name"].lower()))
    doc = {"version": DOC_VERSION, "generated_at": now.isoformat(timespec="seconds"), "data_as_of": scan, "projects": projects}
    validate_no_paths(doc)
    return doc
```

```python
class PublishConfigError(RuntimeError):
    """ACP_PUBLISH_DEST is unset or empty."""


def resolve_dest() -> str:
    dest = (os.environ.get("ACP_PUBLISH_DEST") or "").strip()
    if not dest:
        raise PublishConfigError(
            "ACP_PUBLISH_DEST is not set. Example: "
            "linuxuser@100.106.46.47:/home/astrowithroro.com/public_html/live/shooting.json"
        )
    return dest


def write_document(doc: dict, out_dir: Path) -> Path:
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    target = out_dir / "shooting.json"
    tmp = target.with_suffix(".json.tmp")
    tmp.write_text(json.dumps(doc, indent=2, allow_nan=False), encoding="utf-8")
    os.replace(tmp, target)
    return target


def push_document(path: Path, dest: str, ssh_key: str | None = None) -> subprocess.CompletedProcess:
    ssh_cmd = "ssh -o BatchMode=yes"
    if ssh_key:
        ssh_cmd += f" -i {shlex.quote(ssh_key)}"
    cmd = ["rsync", "-az", "-e", ssh_cmd, str(path), dest]
    return subprocess.run(cmd, capture_output=True, text=True, check=False)
```

```python
#!/usr/bin/env python
"""Publish the public live document to astrowithroro.com/live.

    python scripts/publish_shooting.py --rescan    rebuild the manifest first, then publish
    python scripts/publish_shooting.py             publish from the current manifest
    python scripts/publish_shooting.py --dry-run   write data/live/shooting.json and stop

Destination comes from ACP_PUBLISH_DEST (rsync target), optional key from
ACP_PUBLISH_SSH_KEY. See docs/specs/shooting-page.md and docs/sharing.md.
"""
from __future__ import annotations

import argparse
import os
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--rescan", action="store_true", help="run scripts/build_archive_manifest.py first")
    ap.add_argument("--dry-run", action="store_true", help="write the JSON locally and do not push")
    args = ap.parse_args(argv)

    if args.rescan:
        r = subprocess.run([sys.executable, str(ROOT / "scripts" / "build_archive_manifest.py")])
        if r.returncode != 0:
            print("rescan failed, not publishing", file=sys.stderr)
            return r.returncode

    import shooting_publish as sp
    import app as app_module

    dest = None
    if not args.dry_run:
        try:
            dest = sp.resolve_dest()
        except sp.PublishConfigError as e:
            print(str(e), file=sys.stderr)
            return 2

    plans = app_module.load_plans().get("plans", [])
    doc = sp.build_shooting_document(plans, app_module.load_manifest(), app_module.load_gear())
    path = sp.write_document(doc, app_module.LIVE_OUT_DIR)
    print(f"wrote {path} ({len(doc['projects'])} public project(s))")
    if args.dry_run:
        return 0
    result = sp.push_document(path, dest, os.environ.get("ACP_PUBLISH_SSH_KEY") or None)
    if result.returncode != 0:
        print(result.stderr, file=sys.stderr)
        return result.returncode
    print(f"pushed to {dest}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```javascript
(async function () {
  const status = document.getElementById("live-status");
  const cards = document.getElementById("live-cards");
  let doc;
  try {
    const r = await fetch("shooting.json", { cache: "no-cache" });
    if (!r.ok) throw new Error(r.status);
    doc = await r.json();
    if (!doc || !Array.isArray(doc.projects)) throw new Error("bad shape");
  } catch (e) {
    status.textContent = "Nothing public right now.";
    return;
  }
  if (!doc.projects.length) { status.textContent = "Nothing public right now."; return; }
  const gen = doc.generated_at ? new Date(doc.generated_at) : null;
  status.textContent = `Updated ${gen ? gen.toLocaleDateString("en-AU", { day: "numeric", month: "long" }) : "recently"}` +
    (doc.data_as_of ? `, from a scan on ${new Date(doc.data_as_of + "T00:00:00").toLocaleDateString("en-AU", { day: "numeric", month: "long" })}` : "");

  const esc = s => String(s ?? "").replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  const thumb = p => {
    const fov = Math.max(p.fov_arcmin?.[0] || 60, p.fov_arcmin?.[1] || 60) / 60 * 1.3;
    const q = new URLSearchParams({ hips: "CDS/P/DSS2/color", ra: p.center_ra_deg, dec: p.center_dec_deg, fov: fov.toFixed(3), width: 480, height: 320, projection: "TAN", format: "jpg" });
    return `https://alasky.cds.unistra.fr/hips-image-services/hips2fits?${q}`;
  };
  const nights = p => p.last_imaged == null ? "Not started yet"
    : p.last_imaged_nights_ago === 0 ? "Imaged last night"
    : p.last_imaged_nights_ago === 1 ? "Last imaged 1 night ago"
    : `Last imaged ${p.last_imaged_nights_ago} nights ago`;

  cards.innerHTML = doc.projects.map(p => {
    const bars = Object.entries(p.filters || {}).map(([f, g]) => {
      const pct = g.target_hours > 0 ? Math.min(100, Math.round(100 * g.done_hours / g.target_hours)) : 0;
      return `<div class="bar"><span class="lbl">${esc(f)}</span><span class="track"><span class="fill" style="width:${pct}%"></span></span><span class="num">${g.done_hours} / ${g.target_hours} h</span></div>`;
    }).join("");
    const mosaic = p.mosaic ? ` · ${p.mosaic.rows}×${p.mosaic.cols} mosaic` : "";
    return `<article class="card${p.is_current ? " current" : ""}">
      <img src="${thumb(p)}" alt="" loading="lazy">
      <div class="body">
        ${p.is_current ? '<span class="badge">Current</span>' : ""}
        <h2>${esc(p.project_name)}</h2>
        <p class="meta">${esc(p.target_name)} · ${esc(p.telescope)}${mosaic}</p>
        ${p.blurb ? `<p class="blurb">${esc(p.blurb)}</p>` : ""}
        ${bars}
        <p class="when">${nights(p)}</p>
      </div>
    </article>`;
  }).join("");
})();
```
